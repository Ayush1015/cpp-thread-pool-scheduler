cmake_minimum_required(VERSION 3.16)
project(thread_pool_scheduler VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(ENABLE_TSAN "Enable ThreadSanitizer (GCC/Clang on Linux)" OFF)
add_library(scheduler src/thread_pool.cpp)
target_include_directories(scheduler PUBLIC include)
target_link_libraries(scheduler PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(scheduler PUBLIC -Wall -Wextra -Wpedantic -Werror)
  if(ENABLE_TSAN)
    target_compile_options(scheduler PUBLIC -fsanitize=thread -g -O1)
    target_link_options(scheduler PUBLIC -fsanitize=thread)
  endif()
endif()
add_executable(scheduler_cli src/main.cpp)
target_link_libraries(scheduler_cli PRIVATE scheduler)
include(CTest)
if(BUILD_TESTING)
  add_executable(scheduler_tests tests/test_thread_pool.cpp)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/catch.hpp")
    set(CATCH_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
  else()
    # The zip includes Catch2. A GitHub clone downloads this pinned header once.
    set(CATCH_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/catch2")
    file(MAKE_DIRECTORY "${CATCH_INCLUDE}")
    file(DOWNLOAD
      "https://raw.githubusercontent.com/catchorg/Catch2/v2.13.10/single_include/catch2/catch.hpp"
      "${CATCH_INCLUDE}/catch.hpp"
      EXPECTED_HASH SHA256=3725c0f0a75f376a5005dde31ead0feb8f7da7507644c201b814443de8355170
      TLS_VERIFY ON STATUS CATCH_DOWNLOAD_STATUS)
    list(GET CATCH_DOWNLOAD_STATUS 0 CATCH_DOWNLOAD_CODE)
    if(NOT CATCH_DOWNLOAD_CODE EQUAL 0)
      message(FATAL_ERROR "Catch2 download failed: ${CATCH_DOWNLOAD_STATUS}")
    endif()
  endif()
  target_include_directories(scheduler_tests SYSTEM PRIVATE "${CATCH_INCLUDE}")
  target_link_libraries(scheduler_tests PRIVATE scheduler)
  add_test(NAME scheduler_tests COMMAND scheduler_tests)
  set_tests_properties(scheduler_tests PROPERTIES TIMEOUT 30)
  add_test(NAME cli_demo COMMAND scheduler_cli --demo)
  set_tests_properties(cli_demo PROPERTIES TIMEOUT 10 PASS_REGULAR_EXPRESSION "completed=5")
  add_test(NAME cli_input COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_cli.sh $<TARGET_FILE:scheduler_cli>)
  set_tests_properties(cli_input PROPERTIES TIMEOUT 10)
endif()
